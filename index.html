From npm install to live in 30 seconds.

No infrastructure to provision. No connection limits to negotiate. Subscribe to a channel and start receiving events.

  • ✓ Automatic reconnection with exponential backoff
  • ✓ At-least-once delivery with dedup tokens
  • ✓ Typed payloads via generics
  • ✓ Works in browsers, edge runtimes, and Node
subscribe.ts 
import { Ssma } from "@ssma/client";

const ssma = new Ssma({ apiKey: process.env.SSMA_KEY! });

await ssma
  .channel("orders:live")
  .subscribe<Order>((msg) => {
    console.log("new order", msg.data.id);
  });

// publish from anywhere
await ssma.channel("orders:live").publish({
  id: "ord_42",
  total: 129.00,
});
